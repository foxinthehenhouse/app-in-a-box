table audit_events rls=on
  column id bigint not null generated always as identity
  column actor_id uuid
  column action text not null
  column target text
  column request_id text
  column at timestamp with time zone not null default now()
  constraint audit_events_action_check: CHECK ((action ~ '^[a-z][a-z0-9_]*(\.[a-z][a-z0-9_]*)+$'::text))
  constraint audit_events_pkey: PRIMARY KEY (id)
  constraint audit_events_request_id_check: CHECK ((char_length(request_id) <= 128))
  constraint audit_events_target_check: CHECK ((char_length(target) <= 200))
  index CREATE INDEX audit_events_actor_idx ON public.audit_events USING btree (actor_id, id)
  index CREATE UNIQUE INDEX audit_events_pkey ON public.audit_events USING btree (id)
  trigger CREATE TRIGGER audit_events_no_truncate BEFORE TRUNCATE ON public.audit_events FOR EACH STATEMENT EXECUTE FUNCTION public.audit_events_append_only()
  trigger CREATE TRIGGER audit_events_no_update_delete BEFORE DELETE OR UPDATE ON public.audit_events FOR EACH ROW EXECUTE FUNCTION public.audit_events_append_only()
  grant anon: none
  grant authenticated: none
  grant service_role: select,insert

table push_tokens rls=on
  column id bigint not null generated always as identity
  column user_id uuid not null
  column token text not null
  column platform text
  column created_at timestamp with time zone not null default now()
  column last_seen_at timestamp with time zone not null default now()
  constraint push_tokens_pkey: PRIMARY KEY (id)
  constraint push_tokens_platform_check: CHECK ((platform = ANY (ARRAY['ios'::text, 'android'::text, 'web'::text])))
  constraint push_tokens_token_check: CHECK (((char_length(token) <= 256) AND (token ~ '^Expo(nent)?PushToken\[.+\]$'::text)))
  constraint push_tokens_token_key: UNIQUE (token)
  constraint push_tokens_user_id_fkey: FOREIGN KEY (user_id) REFERENCES auth.users(id) ON DELETE CASCADE
  index CREATE UNIQUE INDEX push_tokens_pkey ON public.push_tokens USING btree (id)
  index CREATE UNIQUE INDEX push_tokens_token_key ON public.push_tokens USING btree (token)
  index CREATE INDEX push_tokens_user_idx ON public.push_tokens USING btree (user_id, last_seen_at DESC)
  policy push_tokens_delete_own: delete to public using (( SELECT auth.uid() AS uid) = user_id)
  policy push_tokens_select_own: select to public using (( SELECT auth.uid() AS uid) = user_id)
  grant anon: select,insert,update,delete
  grant authenticated: select,insert,update,delete
  grant service_role: select,insert,update,delete
